  CREATE OR REPLACE FUNCTION entrasp.MdT_codTitolo(
    codice_azienda text,
    id_mdt numeric)
  RETURNS text AS
$BODY$
SELECT 
codice || ' - ' || titolo
FROM 
  entrasp.modelli_test

WHERE 
  codice_azienda=$1 AND
  id_modello_test=$2
$BODY$
  LANGUAGE sql VOLATILE
  COST 100;

  OWNER TO postgres;
