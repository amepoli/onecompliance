	SONDAGGI */
 select sondaggi.codice_azienda, sondaggi.id_sondaggio, sondaggi.data_comp, sondaggi.data_prevista, sondaggi.data_esecuzione, sondaggi.titolo, entrasp.MdT_codTitolo('DEMO', sondaggi.id_modello_test), entrasp.centri_gestionali_descr('DEMO',sondaggi.id_centro_gest), entrasp.cpl_giudizi_verifiche_descrizione('DEMO', cpl_risultati.id_risultato), sondaggi.stato

 WHERE codice_azienda='DEMO';
